document.addEventListener('DOMContentLoaded', function() { // 设置恋爱开始日期（这里需要手动设置，或者可以让用户设置，但为了简单，我们写死） const startDate = new Date('2023-01-01'); // 修改为你的恋爱开始日期 const today = new Date(); const diffTime = Math.abs(today - startDate); const diffDays = Math.ceil(diffTime / (1000 * 60 * 60 * 24)); document.getElementById('days').textContent = diffDays; document.getElementById('start-date').textContent = startDate.toISOString().split('T')[0]; // 从localStorage加载记录 loadRecords(); // 表单提交事件 document.getElementById('record-form').addEventListener('submit', function(e) { e.preventDefault(); const content = document.getElementById('content').value; const imageFile = document.getElementById('image-upload').files[0]; if (!content && !imageFile) { alert('请至少输入文字或选择一张图片'); return; } const record = { id: Date.now(), date: new Date().toLocaleDateString(), content: content, imageUrl: null }; // 如果有图片，转换为base64保存（注意：localStorage有大小限制，图片不能太大） if (imageFile) { const reader = new FileReader(); reader.onload = function(e) { record.imageUrl = e.target.result; saveRecord(record); loadRecords(); }; reader.readAsDataURL(imageFile); } else { saveRecord(record); loadRecords(); } // 重置表单 document.getElementById('record-form').reset(); }); }); function saveRecord(record) { let records = JSON.parse(localStorage.getItem('loveRecords')) || []; records.unshift(record); // 新记录放在前面 localStorage.setItem('loveRecords', JSON.stringify(records)); } function loadRecords() { const records = JSON.parse(localStorage.getItem('loveRecords')) || []; const recordsList = document.getElementById('records-list'); recordsList.innerHTML = ''; records.forEach(record => { const recordElement = document.createElement('div'); recordElement.className = 'record-item'; recordElement.innerHTML = `
${record.date}
${record.content}
${record.imageUrl ? `` : ''} `; recordsList.appendChild(recordElement); }); }
